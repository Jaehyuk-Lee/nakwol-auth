function node(tag, className, text) {
  const element = document.createElement(tag);
  if (className) element.className = className;
  if (text !== undefined && text !== null) element.textContent = text;
  return element;
}

const IDENTITY_VARIANTS = new Set(['button', 'compact', 'menu']);
const IDENTITY_THEMES = new Set(['inherit', 'light', 'dark']);

function identityRoleLabel(user) {
  const role = user?.membership?.role || 'user';
  if (role === 'admin') return '낙월 관리자';
  if (user?.membership?.is_member || role === 'member') return '낙월 맹원';
  return '일반 사용자';
}

export function mountNakwolIdentityMenu(client, options = {}) {
  if (!(client instanceof NakwolAuthClient)) throw new NakwolAuthError('INVALID_CLIENT', 'NakwolAuthClient 인스턴스가 필요합니다.');

  const variant = options.variant || 'compact';
  const theme = options.theme || 'inherit';
  if (!IDENTITY_VARIANTS.has(variant)) throw new NakwolAuthError('INVALID_VARIANT', `지원하지 않는 Identity Menu variant입니다: ${variant}`);
  if (!IDENTITY_THEMES.has(theme)) throw new NakwolAuthError('INVALID_THEME', `지원하지 않는 Identity Menu theme입니다: ${theme}`);

  ensureIdentityStyles();
  const root = options.container || node('div', 'nakwol-identity');
  root.classList.add('nakwol-identity');
  root.dataset.variant = variant;
  root.dataset.theme = theme;
  if (!options.container) document.body.appendChild(root);

  const showName = options.showName !== false;
  const showRole = options.showRole === true;
  let trigger = null;
  let menu = null;
  let destroyed = false;

  const clear = () => {
    while (root.firstChild) root.removeChild(root.firstChild);
    trigger = null;
    menu = null;
  };

  const accountHref = (hash = '') => {
    const url = new URL(options.accountUrl || `${client.authOrigin}/account`);
    if (!options.accountUrl) url.searchParams.set('client_id', client.clientId);
    if (hash) url.hash = hash;
    return url.toString();
  };

  const setOpen = (open, returnFocus = false) => {
    if (!trigger || !menu) return;
    trigger.setAttribute('aria-expanded', open ? 'true' : 'false');
    menu.hidden = !open;
    if (!open && returnFocus) trigger.focus();
  };

  const renderLoading = () => {
    if (destroyed) return;
    clear();
    root.appendChild(node('span', 'nakwol-identity__status', options.loadingText || '낙월 인증 확인 중…'));
  };

  const renderLoggedOut = (errorMessage = '') => {
    if (destroyed) return;
    clear();
    if (errorMessage) {
      const error = node('span', 'nakwol-identity__status', errorMessage);
      error.dataset.error = 'true';
      root.appendChild(error);
    }
    const login = node('button', 'nakwol-identity__button', options.loginText || '낙월 로그인');
    login.type = 'button';
    login.addEventListener('click', () => client.login());
    root.appendChild(login);
  };

  const renderUser = (user) => {
    if (destroyed) return;
    clear();

    trigger = node('button', 'nakwol-identity__trigger');
    trigger.type = 'button';
    trigger.setAttribute('aria-haspopup', 'menu');
    trigger.setAttribute('aria-expanded', 'false');
    trigger.setAttribute('aria-label', '낙월 계정 메뉴 열기');

    if (user?.avatar_url) {
      const avatar = node('img', 'nakwol-identity__avatar');
      avatar.src = user.avatar_url;
      avatar.alt = '';
      trigger.appendChild(avatar);
    } else {
      const fallback = node('span', 'nakwol-identity__avatar', String(user?.display_name || user?.id || 'N').slice(0, 1).toUpperCase());
      fallback.setAttribute('aria-hidden', 'true');
      trigger.appendChild(fallback);
    }

    if (showName) trigger.appendChild(node('span', 'nakwol-identity__name', user?.display_name || user?.id || 'NAKWOL ID'));
    trigger.appendChild(node('span', 'nakwol-identity__chevron', '▾'));

    menu = node('div', 'nakwol-identity__menu');
    menu.hidden = true;
    menu.setAttribute('role', 'menu');

    const summary = node('div', 'nakwol-identity__summary');
    summary.appendChild(node('div', 'nakwol-identity__summary-name', user?.display_name || user?.id || 'NAKWOL ID'));
    if (showRole) summary.appendChild(node('div', 'nakwol-identity__summary-role', identityRoleLabel(user)));
    menu.appendChild(summary);

    if (options.showAccountLinks === true) {
      const accountLink = node('a', '', '내 낙월 계정');
      accountLink.href = accountHref();
      accountLink.setAttribute('role', 'menuitem');
      menu.appendChild(accountLink);
  
      const permissionLink = node('a', '', '이 서비스 권한');
      permissionLink.href = accountHref('permissions');
      permissionLink.setAttribute('role', 'menuitem');
      menu.appendChild(permissionLink);
    }

    const logout = node('button', '', options.logoutText || '로그아웃');
    logout.type = 'button';
    logout.setAttribute('role', 'menuitem');
    logout.addEventListener('click', async () => {
      setOpen(false);
      await client.logout({ global: Boolean(options.globalLogout) });
    });
    menu.appendChild(logout);

    trigger.addEventListener('click', () => setOpen(trigger.getAttribute('aria-expanded') !== 'true'));
    root.append(trigger, menu);
  };

  const onDocumentPointer = (event) => {
    if (!root.contains(event.target)) setOpen(false);
  };
  const onDocumentKey = (event) => {
    if (event.key === 'Escape' && trigger?.getAttribute('aria-expanded') === 'true') setOpen(false, true);
  };
  document.addEventListener('pointerdown', onDocumentPointer);
  document.addEventListener('keydown', onDocumentKey);

  const onLoading = () => renderLoading();
  const onUser = (event) => renderUser(event.detail);
  const onLogout = () => renderLoggedOut();
  const onError = (event) => renderLoggedOut(`인증 실패: ${event.detail?.code || 'unknown'}`);
  client.addEventListener('loading', onLoading);
  client.addEventListener('user', onUser);
  client.addEventListener('logout', onLogout);
  client.addEventListener('error', onError);

  renderLoading();
  const ready = client.bootstrap().then((user) => {
    if (user) renderUser(user);
    else renderLoggedOut();
    return user;
  }).catch((error) => {
    renderLoggedOut(`인증 실패: ${error?.code || 'unknown'}`);
    throw error;
  });

  return {
    element: root,
    ready,
    refresh: async () => {
      renderLoading();
      const user = await client.getMe();
      if (user) renderUser(user);
      else renderLoggedOut();
      return user;
    },
    destroy: () => {
      if (destroyed) return;
      destroyed = true;
      document.removeEventListener('pointerdown', onDocumentPointer);
      document.removeEventListener('keydown', onDocumentKey);
      client.removeEventListener('loading', onLoading);
      client.removeEventListener('user', onUser);
      client.removeEventListener('logout', onLogout);
      client.removeEventListener('error', onError);
      root.remove();
    },
  };
}
