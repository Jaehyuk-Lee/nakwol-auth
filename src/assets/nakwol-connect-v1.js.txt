(() => {
  const script = document.currentScript;
  if (!script) return;

  const clientId = script.dataset.clientId || '';
  const authOrigin = (script.dataset.authOrigin || new URL(script.src, location.href).origin).replace(/\/$/, '');
  const dataOrigin = (script.dataset.dataOrigin || 'https://nakwol-data.sepsd21.workers.dev').replace(/\/$/, '');
  const dataScopes = [...new Set((script.dataset.dataScopes || '').split(',').map((value) => value.trim()).filter(Boolean))].sort();
  const redirectUri = script.dataset.redirectUri || `${location.origin}${location.pathname}`;
  const uiMode = script.dataset.ui || 'widget';
  const globalLogout = script.dataset.globalLogout === 'true';
  const autoSso = script.dataset.autoSso !== 'false';
  const authMode = (script.dataset.auth || 'required').trim().toLowerCase();

  if (!clientId) { console.error('[NAKWOL Connect] data-client-id is required.'); return; }
  if (!['required', 'optional'].includes(authMode)) { console.error('[NAKWOL Connect] data-auth must be required or optional.'); return; }

  const authRequired = authMode === 'required';
  const guardKey = `__NAKWOL_CONNECT_${clientId}`;
  if (window[guardKey]) return;
  window[guardKey] = true;
  const dispatch = (name, detail) => window.dispatchEvent(new CustomEvent(name, { detail }));

  let activeClient = null;
  let authGuard = null;
  let bodyLock = null;

  const lockBody = () => {
    if (!authRequired || bodyLock || !document.body) return;
    bodyLock = {
      inert: document.body.hasAttribute('inert'),
      overflow: document.documentElement.style.overflow,
    };
    document.body.setAttribute('inert', '');
    document.documentElement.style.overflow = 'hidden';
  };

  const ensureAuthGuard = () => {
    if (!authRequired) return null;
    if (authGuard) { lockBody(); return authGuard; }

    const root = document.createElement('div');
    root.setAttribute('data-nakwol-connect-auth-guard', '');
    root.setAttribute('role', 'dialog');
    root.setAttribute('aria-modal', 'true');
    Object.assign(root.style, {
      position: 'fixed',
      inset: '0',
      zIndex: '2147483646',
      display: 'grid',
      placeItems: 'center',
      padding: '24px',
      background: '#0b0f18',
      color: '#f8fafc',
      fontFamily: 'system-ui,-apple-system,BlinkMacSystemFont,"Segoe UI",sans-serif',
    });

    const card = document.createElement('div');
    Object.assign(card.style, {
      width: 'min(420px, 100%)',
      padding: '24px',
      border: '1px solid #283548',
      borderRadius: '16px',
      background: '#111827',
      boxShadow: '0 24px 70px rgba(0,0,0,.35)',
      textAlign: 'center',
    });
    const title = document.createElement('strong');
    Object.assign(title.style, { display: 'block', fontSize: '18px', lineHeight: '1.4' });
    title.textContent = 'NAKWOL 인증 확인 중…';
    const copy = document.createElement('p');
    Object.assign(copy.style, { margin: '9px 0 0', color: '#aab6c8', fontSize: '13px', lineHeight: '1.6' });
    copy.textContent = '이 서비스는 낙월 인증이 필요합니다.';
    const button = document.createElement('button');
    button.type = 'button';
    button.hidden = true;
    button.textContent = '낙월 로그인';
    Object.assign(button.style, {
      marginTop: '16px',
      border: '0',
      borderRadius: '10px',
      padding: '10px 14px',
      background: '#5865f2',
      color: '#fff',
      fontSize: '13px',
      fontWeight: '750',
      cursor: 'pointer',
    });
    button.addEventListener('click', () => activeClient?.login());
    card.append(title, copy, button);
    root.appendChild(card);
    document.documentElement.appendChild(root);
    authGuard = { root, title, copy, button };
    lockBody();
    return authGuard;
  };

  const setAuthGuardState = (state, error = null) => {
    const guard = ensureAuthGuard();
    if (!guard) return;
    if (state === 'login') {
      guard.title.textContent = '낙월 로그인이 필요합니다';
      guard.copy.textContent = '로그인 화면으로 이동합니다.';
      guard.button.hidden = true;
      return;
    }
    if (state === 'loggedout') {
      guard.title.textContent = '낙월 로그인이 필요합니다';
      guard.copy.textContent = '이 서비스는 낙월 인증 사용자만 이용할 수 있습니다.';
      guard.button.textContent = '낙월 로그인';
      guard.button.hidden = false;
      return;
    }
    if (state === 'error') {
      const code = error?.code || 'unknown';
      const denied = code === 'access_denied' || code === 'ACCESS_DENIED';
      guard.title.textContent = denied ? '접근 권한 없음' : '인증에 실패했습니다';
      guard.copy.textContent = denied
        ? '이 서비스는 허용된 낙월 사용자만 이용할 수 있습니다.'
        : `인증 오류: ${code}`;
      guard.button.textContent = denied ? 'Discord로 권한 다시 확인' : '다시 로그인';
      guard.button.hidden = false;
    }
  };

  const releaseAuthGuard = () => {
    if (!authRequired) return;
    authGuard?.root.remove();
    authGuard = null;
    if (bodyLock && document.body) {
      if (!bodyLock.inert) document.body.removeAttribute('inert');
      document.documentElement.style.overflow = bodyLock.overflow;
    }
    bodyLock = null;
  };

  // 이미 검증된 세션 복원(v1.4): 같은 탭에 유효한 앱 토큰과, 그 토큰으로 /me 검증에 성공한 사용자 캐시가
  // 함께 있으면 인증 화면 없이 먼저 보여 주고 /me 재검증은 뒤에서 한다. 재검증이 실패하면 즉시 다시 잠근다.
  // 캐시는 토큰의 expiresAt에 묶여 있어 다른 토큰(새 로그인·만료 후 재발급)에는 쓰이지 않는다.
  const tokenKey = `nakwol.auth.${clientId}.token`;
  const verifiedUserKey = `nakwol.auth.${clientId}.connect_verified_user`;
  const readTokenRecord = () => {
    try {
      const record = JSON.parse(window.sessionStorage.getItem(tokenKey) || 'null');
      return record?.accessToken && record?.expiresAt && Date.now() < record.expiresAt ? record : null;
    } catch { return null; }
  };
  const readVerifiedUser = () => {
    const record = readTokenRecord();
    if (!record) return null;
    try {
      const cached = JSON.parse(window.sessionStorage.getItem(verifiedUserKey) || 'null');
      return cached?.user && cached.tokenExpiresAt === record.expiresAt ? cached.user : null;
    } catch { return null; }
  };
  const rememberVerifiedUser = (user) => {
    const record = readTokenRecord();
    try {
      if (user && record) window.sessionStorage.setItem(verifiedUserKey, JSON.stringify({ user, tokenExpiresAt: record.expiresAt }));
      else window.sessionStorage.removeItem(verifiedUserKey);
    } catch {}
  };
  const restoredUser = authRequired ? readVerifiedUser() : null;

  if (authRequired && !restoredUser) ensureAuthGuard();
  if (restoredUser) {
    window.NAKWOL_AUTH_USER = restoredUser;
    dispatch('nakwol-session-restored', restoredUser);
  }

  const start = async () => {
    try {
      if (authRequired && !restoredUser) ensureAuthGuard();
      const sdk = await import(`${authOrigin}/sdk/v0.3.1/nakwol-auth-web.js`);
      const client = new sdk.NakwolAuthClient({ clientId, redirectUri, authOrigin, autoSso });
      if (script.dataset.serverGate === 'true') {
        const rememberReturnPage = () => {
          try {
            sessionStorage.setItem(`nakwol:server:return:${clientId}`, JSON.stringify({ path:location.pathname + location.search + location.hash, at:Date.now() }));
          } catch { /* The login bridge falls back to the site root without storage. */ }
        };
        client.addEventListener('ssoprobe', rememberReturnPage);
        client.addEventListener('loginstart', rememberReturnPage);
        const localLogout = client.logout.bind(client);
        client.logout = async (options = {}) => {
          const result = await fetch('/__nakwol/logout', { method:'POST', cache:'no-store', signal:AbortSignal.timeout(10000) });
          if (!result.ok) throw new Error('서버 로그아웃에 실패했습니다. 다시 시도해 주세요.');
          await localLogout(options);
          if (result.headers.get('X-Nakwol-Revoke') === 'failed') dispatch('nakwol-error', { code:'SERVER_REVOKE_FAILED', message:'이 브라우저의 로그아웃은 완료했지만 서버 토큰 폐기를 확인하지 못했습니다.' });
        };
      }
      activeClient = client;

      const enforceRequiredAuth = async (user) => {
        if (!authRequired) return user || null;
        if (user) { releaseAuthGuard(); return user; }
        const suppressed = client.storage.getItem(client.key('sso_suppressed')) === '1';
        if (suppressed) {
          setAuthGuardState('loggedout');
          return null;
        }
        setAuthGuardState('login');
        await client.login();
        return new Promise(() => {});
      };

      class NakwolDataError extends Error {
        constructor(code, message, status = 0, payload = null) { super(message || code); this.name = 'NakwolDataError'; this.code = code; this.status = status; this.payload = payload; }
      }
      const dataFetch = async (path, options = {}) => {
        if (typeof path !== 'string' || !path.startsWith('/')) throw new NakwolDataError('NAKWOL_DATA_INVALID_PATH', 'DATA path는 / 로 시작해야 합니다.');
        const token = client.getAccessToken();
        if (!token) throw new NakwolDataError('NAKWOL_DATA_UNAUTHENTICATED', 'NAKWOL 로그인이 필요합니다.', 401);
        const headers = new Headers(options.headers || {});
        headers.set('Authorization', `Bearer ${token}`);
        headers.set('X-NAKWOL-CLIENT-ID', clientId);
        return fetch(`${dataOrigin}${path}`, { ...options, headers });
      };
      const dataRequest = async (path, options = {}) => {
        const response = await dataFetch(path, options);
        const payload = await response.json().catch(() => ({}));
        if (!response.ok) {
          const error = payload?.error || {};
          throw new NakwolDataError(error.code || 'NAKWOL_DATA_ERROR', error.message || `DATA HTTP ${response.status}`, response.status, payload);
        }
        return payload;
      };
      const dataOpenApi = async () => {
        const response = await fetch(`${dataOrigin}/openapi.json`, { headers: { 'Cache-Control': 'no-cache' } });
        const payload = await response.json().catch(() => ({}));
        if (!response.ok) {
          const error = payload?.error || {};
          throw new NakwolDataError(error.code || 'NAKWOL_DATA_DISCOVERY_ERROR', error.message || `DATA HTTP ${response.status}`, response.status, payload);
        }
        return payload;
      };
      const segment = (value) => encodeURIComponent(String(value));
      const jsonRequest = (path, method, body) => {
        const headers = new Headers({ 'Content-Type': 'application/json' });
        return dataRequest(path, { method, headers, body: JSON.stringify(body) });
      };
      const accountPath = (accountId) => `/v1/game-accounts/${segment(accountId)}`;
      const deckPath = (accountId, deckId) => `${accountPath(accountId)}/decks/${segment(deckId)}`;

      const data = {
        version: '0.4.0', origin: dataOrigin, scopes: dataScopes,
        hasScope: (scope) => dataScopes.includes(scope),
        fetch: dataFetch, request: dataRequest, openapi: dataOpenApi, describe: dataOpenApi,
        accounts: {
          list: () => dataRequest('/v1/game-accounts'),
          create: (input) => jsonRequest('/v1/game-accounts', 'POST', input),
        },
        roster: {
          generals: {
            list: (accountId) => dataRequest(`${accountPath(accountId)}/roster/generals`),
            upsert: (accountId, generalId, input) => jsonRequest(`${accountPath(accountId)}/roster/generals/${segment(generalId)}`, 'PUT', input),
            remove: (accountId, generalId) => dataRequest(`${accountPath(accountId)}/roster/generals/${segment(generalId)}`, { method: 'DELETE' }),
          },
          tactics: {
            list: (accountId) => dataRequest(`${accountPath(accountId)}/roster/tactics`),
            upsert: (accountId, tacticId, input) => jsonRequest(`${accountPath(accountId)}/roster/tactics/${segment(tacticId)}`, 'PUT', input),
            remove: (accountId, tacticId) => dataRequest(`${accountPath(accountId)}/roster/tactics/${segment(tacticId)}`, { method: 'DELETE' }),
          },
        },
        equipment: {
          list: (accountId) => dataRequest(`${accountPath(accountId)}/equipment`),
          create: (accountId, input) => jsonRequest(`${accountPath(accountId)}/equipment`, 'POST', input),
          update: (accountId, equipmentId, patch) => jsonRequest(`${accountPath(accountId)}/equipment/${segment(equipmentId)}`, 'PATCH', patch),
          remove: (accountId, equipmentId) => dataRequest(`${accountPath(accountId)}/equipment/${segment(equipmentId)}`, { method: 'DELETE' }),
        },
        decks: {
          list: (accountId) => dataRequest(`${accountPath(accountId)}/decks`),
          get: (accountId, deckId) => dataRequest(deckPath(accountId, deckId)),
          create: (accountId, input) => jsonRequest(`${accountPath(accountId)}/decks`, 'POST', input),
          update: (accountId, deckId, patch) => jsonRequest(deckPath(accountId, deckId), 'PATCH', patch),
          replaceComposition: (accountId, deckId, composition) => jsonRequest(`${deckPath(accountId, deckId)}/composition`, 'PUT', composition),
          remove: (accountId, deckId) => dataRequest(deckPath(accountId, deckId), { method: 'DELETE' }),
        },
        snapshots: {
          list: () => dataRequest('/v1/deck-snapshots'),
          get: (snapshotId) => dataRequest(`/v1/deck-snapshots/${segment(snapshotId)}`),
          create: (accountId, deckId, input) => jsonRequest(`${deckPath(accountId, deckId)}/snapshots`, 'POST', input),
        },
        registry: {
          summary: () => dataRequest('/v1/registry/summary'),
          generals: (options = {}) => dataRequest(`/v1/registry/generals${options.includeHidden ? '?include_hidden=1' : ''}`),
          tactics: () => dataRequest('/v1/registry/tactics'),
          equipment: () => dataRequest('/v1/registry/equipment'),
          equipmentTraits: () => dataRequest('/v1/registry/equipment-traits'),
          stats: () => dataRequest('/v1/registry/stats'),
          formations: () => dataRequest('/v1/registry/formations'),
          warbooks: () => dataRequest('/v1/registry/warbooks'),
        },
      };
      const api = {
        version: '1.4.0', clientId, redirectUri, authOrigin, authMode, authRequired, autoSso, client, data, user: null, restoredUser,
        login: () => client.login(), logout: (options = {}) => client.logout(options), getMe: () => client.getMe(),
        isAuthenticated: () => client.isAuthenticated(), isMember: () => client.isMember(),
        getAccessToken: () => client.getAccessToken(),
      };
      window.NAKWOL_CONNECT = api;
      window.NAKWOL_AUTH = client;
      window.NAKWOL_DATA = data;
      client.addEventListener('user', (event) => {
        api.user = event.detail || null;
        rememberVerifiedUser(api.user);
        window.NAKWOL_AUTH_USER = api.user;
        if (api.user && authRequired) releaseAuthGuard();
        dispatch('nakwol-ready', api.user);
        dispatch('nakwol-auth-ready', api.user);
      });
      client.addEventListener('logout', () => {
        api.user = null;
        rememberVerifiedUser(null);
        delete window.NAKWOL_AUTH_USER;
        if (authRequired) setAuthGuardState('loggedout');
        dispatch('nakwol-logout', null);
        dispatch('nakwol-auth-logout', null);
      });
      client.addEventListener('error', (event) => {
        rememberVerifiedUser(null);
        api.user = null;
        delete window.NAKWOL_AUTH_USER;
        if (authRequired) setAuthGuardState('error', event.detail || null);
        dispatch('nakwol-error', event.detail || null);
      });
      if (uiMode === 'none' || uiMode === 'headless') {
        const user = await enforceRequiredAuth(await client.bootstrap());
        api.user = user || null;
        if (user) window.NAKWOL_AUTH_USER = user;
        dispatch('nakwol-ready', api.user);
        dispatch('nakwol-auth-ready', api.user);
        return;
      }
      const widget = sdk.mountNakwolIdentityMenu(client, {
        globalLogout,
        variant: 'compact',
        theme: 'dark',
      });
      Object.assign(widget.element.style, {
        position: 'fixed',
        top: '14px',
        right: '14px',
        zIndex: '2147483000',
      });
      api.widget = widget;
      const user = await enforceRequiredAuth(await widget.ready);
      api.user = user || null;
      if (user) window.NAKWOL_AUTH_USER = user;
      dispatch('nakwol-ready', api.user);
      dispatch('nakwol-auth-ready', api.user);
    } catch (error) {
      rememberVerifiedUser(null);
      if (authRequired) setAuthGuardState('error', error);
      console.error('[NAKWOL Connect] bootstrap failed', error);
      dispatch('nakwol-error', error);
    }
  };
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', start, { once: true }); else start();
})();
